cmake_minimum_required(VERSION 3.20)
project(leogie_preview C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/.local/lvgl-preview" lvgl)
set(BOARD "${ROOT}/main/boards/ai-passport-leogie")
add_executable(preview preview.cc "${BOARD}/leogie_display.cc" "${BOARD}/leogie_art.c" "${BOARD}/leogie_font20.c" "${BOARD}/leogie_digits48.c")
target_include_directories(preview PRIVATE stubs "${BOARD}")
target_compile_definitions(preview PRIVATE BUILTIN_TEXT_FONT=lv_font_montserrat_20)
target_link_libraries(preview PRIVATE lvgl z)
